agregate function - for only numeric field
sum
    find total amount of payment
    SELECT SUM(amount) FROM payments
    SELECT SUM(amount) as 'total amount' FROM payments

avg
    find average amount of payment
    SELECT AVG(amount) as 'average amount' FROM payments

count
    find total number of customers
    SELECT COUNT(*|<fieldname>) as "total customers" FROM customers

    find number of customers who is from usa
    SELECT count(*) AS 'USA customers' FROM customers WHERE country='USA'

    find number of customers who is not from usa,india
    SELECT COUNT(*) FROM customers WHERE country not in('USA','India')
    
min
    find minimum amount from payment
    select min(amount) as "min payment" from payments

max
    find maximum amount from payment
    select max(amount) as "max payment" from payments

find sum,avg,count,min,max of customers
SELECT SUM(creditLimit) AS "total",AVG(creditLimit) AS "average",count(*) AS "count",MIN(creditLimit) as "min",MAX(creditLimit) AS "max" FROM customers;

group by and having
used for gouping

find each customer total payment
SELECT customerNumber,SUM(amount) as total  FROM payments GROUP BY customerNumber

find count of customer that done payment
select customerNumber,count(*) as count from payments group by customerNumber

find those that make payment in year 2004
SELECT paymentDate,customerNumber FROM payments WHERE paymentDate>='2004-01-01' AND paymentDate<='2004-12-31';

having condition
find those country which have less than 3 customers
SELECT country,COUNT(*) FROM customers GROUP BY country HAVING COUNT(*)<4 ORDER BY country
